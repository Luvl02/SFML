cmake_minimum_required(VERSION 3.15)
project(SFML VERSION 1.0 LANGUAGES CXX)


# Force C++17 standard matching your original tasks.

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Automatically find SFML on your Mac via Homebrew
list(APPEND CMAKE_PREFIX_PATH "/opt/homebrew")
find_package(SFML 3 COMPONENTS Graphics Window System REQUIRED)

include_directories("/opt/homebrew/include")

# Dynamically gather ALL .cpp files in your directory automatically
set(SOURCES
src/main.cpp
src/Player.h
src/Player.cpp
src/Layer.h
src/Layer.cpp
src/Enemy.h
src/Enemy.cpp
src/Sanzoku.h
src/Sanzoku.cpp
)

# Define your executable and its source files
add_executable(${PROJECT_NAME} ${SOURCES})

# Link SFML libraries and include paths to all of your code files
target_link_libraries(${PROJECT_NAME} PRIVATE SFML::Graphics SFML::Window SFML::System)

add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets"
    "$<TARGET_FILE_DIR:${PROJECT_NAME}>/assets"
)
